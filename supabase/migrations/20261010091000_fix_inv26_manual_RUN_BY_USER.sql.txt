  ret uuid := '257eda9d-9791-45fd-8473-a8b44f427dde';
  st  uuid := 'a15f496a-a4b5-4388-b304-5397eaeeda1b';
  cus uuid := 'deb63dad-4378-42a8-9dcc-2d8a462d3b2d';
  stf uuid := 'c54a581a-5ce2-4aa4-9942-2873365698b0';
  p100 uuid := 'ec01af07-88aa-4998-ade9-5638d2d326c3';

begin

  update invoices set status = 'unpaid', payment_method = 'mixed', total_amount = 1326.15, amount_paid = 796.15,
         cash_paid_amount = 796.15, bank_paid_amount = 0, pay_currency = null, pay_fx_rate = null,
         pay_breakdown = '[{"currency":"SYP","amount":10350000,"rate":13000.06280223576,"method":"cash"}]'::jsonb
   where id = inv;
  update invoice_items set line_total = 1326.15 where invoice_id = inv;

  -- 2) مسح كل حركات الصندوق والديون القديمة للفاتورة وللمرتجع الملغى
  delete from daily_cash_log where bank_invoice_id in (inv, ret)
     or cash_movement_id in (select id from cash_movements where invoice_id in (inv, ret));
  delete from cash_movements where invoice_id in (inv, ret);
  delete from customer_debts where invoice_id in (inv, ret);
  delete from invoice_items where invoice_id = ret;
  delete from invoices where id = ret;

  -- 3) القبض يوم 5/10: 10,350,000 ل.س نقداً (يدخل الصندوق اليومي بالليرة)
  insert into cash_movements (store_id, box, direction, amount, invoice_id, created_by, description, created_at)

  -- 4) الدين يوم 8/10 (يوم التبديل): 530$

  update pieces set status = 'available' where id = p118;
  delete from piece_movements where piece_id = p100 and event_type <> 'created';

   where piece_id = p118 and note like 'مرتجع RET-2026-000001%';
